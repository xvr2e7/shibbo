GETTING STARTED

Up and running in 60 seconds

Grab the package, paste the template, and start visualizing.

# Install via npm
npm install pulsegraph

// Create your first interactive visualizer
import { PulseChart } from 'pulsegraph';

const chart = new PulseChart('#container', {
  type: 'spline',
  data: [
    { time: '00:00', value: 24 },
    { time: '04:00', value: 48 },
    { time: '08:00', value: 32 },
    { time: '12:00', value: 95 },
    { time: '16:00', value: 67 },
    { time: '20:00', value: 112 }
  ],
  theme: 'midnight',
  animate: true
});

chart.render();
