app.ts — your first chart

        
// 1 · Install
$ npm install @aurora/charts

// 2 · Import & create
import { LineChart } from '@aurora/charts'
import '@aurora/charts/theme-aurora.css'

const chart = new LineChart({
  el: '#chart',
  data: {
    labels: ['Mon','Tue','Wed','Thu','Fri'],
    series: [
      { name: 'Signups', values: [120, 190, 160, 240, 310] },
      { name: 'Trials',  values: [80, 120, 140, 180, 230] }
    ]
  },
  smooth: true,
  theme: 'aurora'
})

// 3 · Update live, anytime
chart.push({ label: 'Sat', values: [380, 290] })
