Simple API

Get started in minutes with our intuitive API

JavaScript TypeScript React 
import { ChartWave } from 'chartwave';

// Create a beautiful line chart
const chart = new ChartWave('#chart', {
  type: 'line',
  data: {
    labels: ['Jan', 'Feb', 'Mar', 'Apr', 'May'],
    datasets: [{
      label: 'Revenue',
      data: [30, 45, 35, 50, 65],
      color: '#6366f1'
    }]
  },
  options: {
    animations: true,
    responsive: true
  }
});
