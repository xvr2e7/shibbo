Quickstart

Up and charting in 30 seconds.

A tiny setup, a little data, and you’re on your way.

terminal & app.js 
# Add Prism to your project
npm install @prism-charts/core

// Create your first chart
import { lineChart } from '@prism-charts/core';

lineChart(document.querySelector('#chart'), {
  data: [{ month: 'Jan', value: 24 }, ...],
  x: 'month',
  y: 'value'
});
