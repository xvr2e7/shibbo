Read the full guide
app.js 
import { Chartify } from 'novacharts';

const ctx = document.getElementById('myChart');

new Chartify(ctx, {
  type: 'line',
  data: {
    labels: ['Jan', 'Feb', 'Mar', 'Apr', 'May'],
    datasets: [{
      label: 'Sales Data',
      data: [65, 59, 80, 81, 56],
      borderColor: '#58a6ff',
      borderWidth: 2,
      fill: true
    }]
  },
  options: {
    responsive: true,
    theme: 'dark'
  }
});
